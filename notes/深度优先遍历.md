# 深度优先遍历

类似前序遍历

```C
#include <stdio.h>

typedef char VertexType; //顶点
typedef int EdgeType; //边

#define MAXSIZE 100

typedef struct{
    VertexType vertex[MAXSIZE]; //保存顶点
    EdgeType arc[MAXSIZE][MAXSIZE]; //邻接矩阵
    int vertex_num; //顶点数量
    int edge_num; //边数量
}Mat_Graph;

int visited[MAXSIZE]; //记录哪个顶点被访问过

void create_graph(Mat_Graph* G){
    G -> vertex_num = 9;
    G -> edge_num = 15;
    G -> vertex[0] = 'A';
    G -> vertex[1] = 'B';
    G -> vertex[2] = 'C';
    G -> vertex[3] = 'D';
    G -> vertex[4] = 'E';
    G -> vertex[5] = 'F';
    G -> vertex[6] = 'G';
    G -> vertex[7] = 'H';
    G -> vertex[8] = 'I';

    for(int i = 0; i < G -> vertex_num; i++){
        for(int j = 0; j < G -> vertex_num; j++){
            G -> arc[i][j] = 0;
        }
    }

    //A-B A-F
    G -> arc[0][1] = 1;
    G -> arc[0][5] = 1;

    //B-C B-G B-I
    G -> arc[1][2] = 1;
    G -> arc[1][6] = 1;
    G -> arc[1][8] = 1;

    //C-D C-I
    G -> arc[2][3] = 1;
    G -> arc[2][8] = 1;

    //D-E D-G D-H D-I
    G -> arc[3][4] = 1;
    G -> arc[3][6] = 1;
    G -> arc[3][7] = 1;
    G -> arc[3][8] = 1;

    //E-F E-H
    G -> arc[4][5] = 1;
    G -> arc[4][7] = 1;

    //F-G
    G -> arc[5][6] = 1;

    //G-H
    G -> arc[6][7] = 1;

    for(int i = 0; i < G -> vertex_num; i++){
        for(int j = 0; j < G -> vertex_num; j++){
            G -> arc[j][i] = G -> arc[i][j];
        }
    }
}

//深度优先
void dfs(Mat_Graph G, int i){
    visited[i] = 1;
    printf("%c\n", G.vertex[i]);

    for(int j = 0; j < G.vertex_num; j++){
        if(G.arc[i][j] == 1 && visited[j] == 0){
            dfs(G, j);
        }
    }
}

int main(){
    Mat_Graph G;
    create_graph(&G);

    for(int i = 0; i < G.vertex_num; i++){
        visited[i] = 0;
    }
    dfs(G, 0);
    return 0;
}
```
